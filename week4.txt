function leapyear(year) {
    // Return true if the year is divisible by 4 but not divisible by 100 unless it's also divisible by 400
    return (year % 100 === 0) ? (year % 400 === 0) : (year % 4 === 0);
}

document.addEventListener("DOMContentLoaded", () => {
  document.querySelectorAll(".snippet").forEach((snippet) => {
    const textElement = snippet.querySelector("code"); 
    const showButton = snippet.querySelector(".byuttton button:nth-child(1)"); 
    const hideButton = snippet.querySelector(".byuttton button:nth-child(2)"); 
    const copyButton = snippet.querySelector(".byuttton button:nth-child(3)");

    if (!textElement || !showButton || !hideButton || !copyButton) {
      console.error("One or more elements missing in a .snippet section");
      return;
    }

    textElement.style.display = "none";

    showButton.addEventListener("click", () => {
      textElement.style.display = "block";
    });

    hideButton.addEventListener("click", () => {
      textElement.style.display = "none";
    });

    copyButton.addEventListener("click", () => {
      if (textElement.style.display !== "none") {
        const textToCopy = textElement.innerText.trim(); 

        const tempInput = document.createElement("textarea");
        tempInput.value = textToCopy;
        document.body.appendChild(tempInput);
        tempInput.select();
        document.execCommand("copy");
        document.body.removeChild(tempInput);

        alert("Code copied to clipboard!");
      } else {
        alert("Code is hidden! Click 'Show Code' first.");
      }
    });
  });
});
